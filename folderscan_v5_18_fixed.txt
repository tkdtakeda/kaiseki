public class DatInfo {

public static class DatTool {

    static int ReadFull(Stream s, byte[] b, int len) {
        int t = 0;
        while (t < len) { int r = s.Read(b, t, len - t); if (r <= 0) break; t += r; }
        return t;

    static FileStream Open(string p) {

    static bool Sw(byte[] b, int n, params int[] m) {

        if (Sw(b, n, 0xFF, 0xD8, 0xFF)) return "JPEG";
        if (Sw(b, n, 0x89, 0x50, 0x4E, 0x47)) return "PNG";
        if (Sw(b, n, 0x49, 0x49, 0x2A, 0x00)) return "TIFF (LE)";

        if (Sw(b, n, 0x3C, 0x3F, 0x78, 0x6D, 0x6C)) return "XML";

    // self-similarity at lag k : ratio of b[i]==b[i+k]
    static string RecLen(byte[] b, int n, int maxLag, long size, out int bestLen, out double bestScore) {
        bestLen = 0; bestScore = 0;
        int lim = Math.Min(maxLag, n / 2);
        if (lim < 2) return "-";
        double[] sc = new double[lim + 1];
        for (int k = 2; k <= lim; k++) {
            int m = 0; int len = n - k;
            for (int i = 0; i < len; i++) if (b[i] == b[i + k]) m++;
            sc[k] = (double)m / len;
        }
        List<int> ks = new List<int>();
        for (int k = 2; k <= lim; k++) ks.Add(k);
        ks.Sort(delegate (int a, int c) { int r = sc[c].CompareTo(sc[a]); return r != 0 ? r : a.CompareTo(c); });
        int best = ks[0]; int bs = best;
        for (int k = 2; k <= best; k++) { if (sc[k] >= sc[best] * 0.95) { bs = k; break; } }
        bestLen = bs; bestScore = sc[bs] * 100;
        StringBuilder s = new StringBuilder();
        s.Append("base=").Append(bs).Append(" (").Append((sc[bs] * 100).ToString("0.0")).Append("%)");
        if (size % bs == 0) s.Append("(div)");
        s.Append(" | top:");
        for (int i = 0; i < Math.Min(5, ks.Count); i++) {
            int k = ks[i];
            s.Append(' ').Append(k).Append('=').Append((sc[k] * 100).ToString("0.0")).Append('%');
            if (size % k == 0) s.Append("(div)");
        }
        return s.ToString();
    }
    public static DatInfo Analyze(string path, int sampleMax, int maxLag) {
        DatInfo d = new DatInfo();
        long[] cnt = new long[256];
        byte[] sample = new byte[Math.Max(sampleMax, 16)];
        int sn = 0; long n = 0;
        using (SHA256 sha = SHA256.Create())
        using (FileStream fs = Open(path)) {
            byte[] buf = new byte[1048576]; int r;
            while ((r = ReadFull(fs, buf, buf.Length)) > 0) {
                sha.TransformBlock(buf, 0, r, null, 0);
                for (int i = 0; i < r; i++) cnt[buf[i]]++;
                if (sn < sample.Length) { int c = Math.Min(r, sample.Length - sn); Buffer.BlockCopy(buf, 0, sample, sn, c); sn += c; }
                n += r;
                if (r < buf.Length) break;
            }
            sha.TransformFinalBlock(new byte[0], 0, 0);
            d.Sha256 = BitConverter.ToString(sha.Hash).Replace("-", "");
        }
        d.Size = n;
        long pr = cnt[9] + cnt[10] + cnt[13];
        for (int i = 32; i < 127; i++) pr += cnt[i];
        long hi = 0;
        for (int i = 128; i < 256; i++) hi += cnt[i];
        double e = 0;
        if (n > 0) { for (int i = 0; i < 256; i++) { if (cnt[i] > 0) { double p = (double)cnt[i] / n; e -= p * Math.Log(p, 2); } } }
        d.Printable = n > 0 ? 100.0 * pr / n : 0;
        d.Zero = n > 0 ? 100.0 * cnt[0] / n : 0;
        d.High = n > 0 ? 100.0 * hi / n : 0;
        d.Entropy = e; d.LF = cnt[10]; d.CR = cnt[13];
        StringBuilder hh = new StringBuilder();
        for (int i = 0; i < Math.Min(sn, 16); i++) { if (i > 0) hh.Append(' '); hh.Append(HX[sample[i]]); }
        d.HeadHex = hh.ToString();
        d.Magic = Magic(sample, sn);
        if (n == 0) d.Type = "empty";
        else if (d.Printable >= 95.0) d.Type = "text (ASCII)";
        else if (cnt[0] == 0 && d.Printable + d.High >= 98.0 && cnt[10] > 0) d.Type = "text (MBCS/Shift-JIS?)";
        else if (e > 7.5) d.Type = "binary (high entropy: compressed/encrypted/image?)";
        else d.Type = "binary";
        int bl; double bsc;
        d.RecLen = RecLen(sample, sn, maxLag, n, out bl, out bsc);
        d.BestLen = bl; d.BestScore = bsc;
        return d;
    }

    public static List<Unit> Units(string p, bool text, int cp, long maxBytes, int minLen) {
        List<Unit> u = new List<Unit>();
        if (text) {
            using (StreamReader sr = new StreamReader(Open(p), Encoding.GetEncoding(cp), true)) {
                string line; long ln = 0; long read = 0;

                    Unit x = new Unit(); x.Loc = "L" + ln; x.Text = line; u.Add(x);
                    if (maxBytes > 0 && read >= maxBytes) break;
                }
            }
        } else {
            using (FileStream fs = Open(p)) {
                byte[] buf = new byte[1048576]; int r; long off = 0; long start = -1;
                StringBuilder sb = new StringBuilder();
                while ((r = ReadFull(fs, buf, buf.Length)) > 0) {
                    for (int i = 0; i < r; i++) {

                    if (maxBytes > 0 && off >= maxBytes) break;
                    if (r < buf.Length) break;
                }
                Flush(u, sb, start, minLen);
            }
        }
        return u;

    public static List<Field> Profile(List<byte[]> recs, int cp) {
        List<Field> fs = new List<Field>();
        if (recs.Count < 2) return fs;
        int R = recs[0].Length; int N = recs.Count;
        string[] cls = new string[R];
        for (int k = 0; k < R; k++) {
            byte f = recs[0][k]; bool cst = true; int dg = 0, pr = 0, hi = 0;
            foreach (byte[] r in recs) {
                byte v = r[k];
                if (v != f) cst = false;
                if ((v >= 48 && v <= 57) || v == 32 || v == 46 || v == 45 || v == 43) dg++;
                if (v >= 32 && v < 127) pr++; else if (v >= 128) hi++;
            }
            if (cst) cls[k] = (f >= 32 && f < 127) ? "CONST" : "CONSTB";
            else if (dg >= N * 0.98) cls[k] = "NUM";
            else if (pr >= N * 0.98) cls[k] = "TEXT";
            else if (pr + hi >= N * 0.98) cls[k] = "MBCS";
            else cls[k] = "BIN";
        }
        Encoding e = Encoding.GetEncoding(cp);
        int st = 0;
        for (int k = 1; k <= R; k++) {
            if (k == R || cls[k] != cls[st]) {
                Field fd = new Field(); fd.Start = st; fd.Len = k - st; fd.Cls = cls[st];
                HashSet<string> d = new HashSet<string>(); List<string> ex = new List<string>();
                foreach (byte[] r in recs) {
                    string v = Val(r, fd, e);
                    if (d.Count < 100000) d.Add(v);
                    if (ex.Count < 3 && !ex.Contains(v)) ex.Add(v);
                }
                fd.Distinct = d.Count; fd.Ex = Clean(string.Join(" | ", ex.ToArray()));
                fs.Add(fd); st = k;
            }
        }
        return fs;
    }
    static string Val(byte[] r, Field f, Encoding e) {
        if (f.Cls == "BIN" || f.Cls == "CONSTB") {

            else if (f.Len == 8) sb.Append(" (LE=").Append(BitConverter.ToInt64(r, f.Start)).Append(")");
            return sb.ToString();
        }
        return "[" + e.GetString(r, f.Start, f.Len) + "]";
    }

public class KeyIndex {
    Dictionary<string, List<int>> map = new Dictionary<string, List<int>>(StringComparer.Ordinal);
    public int KeyCount { get { return map.Count; } }
    public void Add(int no, HashSet<string> keys) {
        foreach (string k in keys) {
            List<int> l;
            if (!map.TryGetValue(k, out l)) { l = new List<int>(); map[k] = l; }
            if (l.Count == 0 || l[l.Count - 1] != no) l.Add(no);
        }
    }
    public List<string[]> Pairs(int maxFiles, int nEx) {
        Dictionary<long, int> cnt = new Dictionary<long, int>();
        Dictionary<long, List<string>> ex = new Dictionary<long, List<string>>();
        foreach (KeyValuePair<string, List<int>> kv in map) {
            List<int> l = kv.Value;
            if (l.Count < 2 || l.Count > maxFiles) continue;
            for (int i = 0; i < l.Count; i++) for (int j = i + 1; j < l.Count; j++) {
                long pk = ((long)l[i] << 32) | (uint)l[j];
                int c; cnt.TryGetValue(pk, out c); cnt[pk] = c + 1;
                List<string> e; if (!ex.TryGetValue(pk, out e)) { e = new List<string>(); ex[pk] = e; }
                if (e.Count < nEx) e.Add(kv.Key);
            }
        }
        List<string[]> rows = new List<string[]>();
        foreach (KeyValuePair<long, int> kv in cnt)
            rows.Add(new string[] { (kv.Key >> 32).ToString(), (kv.Key & 0xFFFFFFFFL).ToString(), kv.Value.ToString(), string.Join(" ", ex[kv.Key].ToArray()) });
        rows.Sort(delegate (string[] a, string[] b) { return int.Parse(b[2]).CompareTo(int.Parse(a[2])); });
        return rows;
    }
    public List<string[]> Common(int top) {
        List<KeyValuePair<string, List<int>>> l = new List<KeyValuePair<string, List<int>>>();
        foreach (KeyValuePair<string, List<int>> kv in map) if (kv.Value.Count >= 2) l.Add(kv);
        l.Sort(delegate (KeyValuePair<string, List<int>> a, KeyValuePair<string, List<int>> b) { return b.Value.Count.CompareTo(a.Value.Count); });
        List<string[]> rows = new List<string[]>();
        for (int i = 0; i < Math.Min(top, l.Count); i++) {
            List<string> ids = new List<string>();
            for (int j = 0; j < Math.Min(30, l[i].Value.Count); j++) ids.Add(l[i].Value[j].ToString());
            if (l[i].Value.Count > 30) ids.Add("..");
            rows.Add(new string[] { l[i].Key, l[i].Value.Count.ToString(), string.Join(",", ids.ToArray()) });
        }
        return rows;
    }

    public static string BaseName(string s) {
        if (s == null) return "";
        string t = s.Trim().Trim('"', '\'', '`');
        t = Regex.Replace(t, @"%~[A-Za-z]*[0-9]", "");
        t = Regex.Replace(t, @"%[^%\s]+%", "");
        t = Regex.Replace(t, @"\$\{[^}]*\}|\$\([^)]*\)|\$[A-Za-z_]\w*", "");
        int q = t.IndexOfAny(new char[] { '?', '#' }); if (q >= 0) t = t.Substring(0, q);
        t = t.Replace("::", "/");
        int k = Math.Max(t.LastIndexOf('/'), t.LastIndexOf('\\'));
        if (k >= 0) t = t.Substring(k + 1);
        return t.Trim();
    }

    public static List<string[]> Symbols(List<Unit> u, string lang) {
        List<string[]> o = new List<string[]>();
        foreach (Unit x in u) {
            try {
                string s = x.Text; Match m;
                if (lang == "py") {
                    m = PyDef.Match(s); if (m.Success) { o.Add(new string[] { m.Groups[1].Length > 0 ? "method/inner-def" : "def", m.Groups[2].Value, x.Loc }); continue; }
                    m = PyClass.Match(s); if (m.Success) { o.Add(new string[] { "class", m.Groups[2].Value, x.Loc }); continue; }
                    m = PyVar.Match(s); if (m.Success) { string nm = m.Groups[1].Value; o.Add(new string[] { nm == nm.ToUpperInvariant() ? "const" : "var", nm, x.Loc }); continue; }
                } else if (lang == "js" || lang == "html") {
                    foreach (Match k in JsFunc.Matches(s)) o.Add(new string[] { "function", k.Groups[1].Value, x.Loc });
                    foreach (Match k in JsArrow.Matches(s)) o.Add(new string[] { "function", k.Groups[1].Value, x.Loc });
                    foreach (Match k in JsClass.Matches(s)) o.Add(new string[] { "class", k.Groups[1].Value, x.Loc });
                    m = JsTop.Match(s); if (m.Success && !JsArrow.IsMatch(s)) o.Add(new string[] { "top-var", m.Groups[1].Value, x.Loc });
                } else if (lang == "bat") {
                    m = BatLabel.Match(s); if (m.Success) o.Add(new string[] { "label", m.Groups[1].Value, x.Loc });
                } else if (lang == "ps") {
                    m = PsFunc.Match(s); if (m.Success) o.Add(new string[] { "psfunction", m.Groups[1].Value, x.Loc });
                } else if (lang == "cs" || lang == "java") {
                    foreach (Match k in CsType.Matches(s)) o.Add(new string[] { "type:" + k.Groups[1].Value, k.Groups[2].Value, x.Loc });
                    m = CsMethod.Match(s); if (m.Success) o.Add(new string[] { "method", m.Groups[1].Value, x.Loc });
                } else if (lang == "vb") {
                    m = VbProc.Match(s); if (m.Success) o.Add(new string[] { m.Groups[1].Value.ToLowerInvariant().Replace(" ", "-"), m.Groups[2].Value, x.Loc });
                } else if (lang == "rs") {
                    m = RsItem.Match(s); if (m.Success) o.Add(new string[] { m.Groups[1].Value, m.Groups[2].Value, x.Loc });
                } else if (lang == "sql") {
                    foreach (Match k in SqlDef.Matches(s)) o.Add(new string[] { "sql:" + k.Groups[1].Value.ToLowerInvariant(), k.Groups[2].Value, x.Loc });
                }
            } catch (RegexMatchTimeoutException) { Tmo(x.Loc); }
        }
        return o;
    }
    public static bool HasMain(List<Unit> u, string lang) {
        Regex r = null;
        if (lang == "py") r = PyMain; else if (lang == "rs") r = RsMain; else if (lang == "cs") r = CsMain; else if (lang == "java") r = JavaMain;
        if (r == null) return false;
        foreach (Unit x in u) { try { if (r.IsMatch(x.Text)) return true; } catch (RegexMatchTimeoutException) { Tmo(x.Loc); } }
        return false;
    }

    static bool SkipHref(string v) {
        string t = v.Trim().ToLowerInvariant();
        return t.Length == 0 || t.Contains("{{") || t.Contains("{%") || t.StartsWith("http:") || t.StartsWith("https:") || t.StartsWith("//") || t.StartsWith("data:") || t.StartsWith("javascript:") || t.StartsWith("mailto:") || t == "#";
    }
    public static List<string[]> Imports(List<Unit> u, string lang) {
        List<string[]> o = new List<string[]>();
        foreach (Unit x in u) {
            try {
                string s = x.Text; Match m;
                if (lang == "py") {
                    m = PyFrom.Match(s);
                    if (m.Success) {
                        o.Add(new string[] { "from", m.Groups[1].Value, m.Groups[2].Value.Replace("(", "").Replace(")", "").Trim(), x.Loc });
                    } else {
                        m = PyImp.Match(s);
                        if (m.Success) {
                            foreach (string part in m.Groups[1].Value.Split(',')) {
                                string p = part.Trim(); if (p.Length == 0) continue;
                                string alias = ""; int a = p.IndexOf(" as ");
                                if (a > 0) { alias = "as " + p.Substring(a + 4).Trim(); p = p.Substring(0, a).Trim(); }
                                if (Regex.IsMatch(p, @"^[\w\.]+$")) o.Add(new string[] { "import", p, alias, x.Loc });
                            }
                        }
                    }
                    foreach (Match k in Tpl.Matches(s)) o.Add(new string[] { "template", k.Groups[1].Value, "", x.Loc });
                } else if (lang == "js") {
                    foreach (Match k in JsImp.Matches(s)) o.Add(new string[] { "import", k.Groups[1].Value, "", x.Loc });
                    foreach (Match k in JsReq.Matches(s)) o.Add(new string[] { "require", k.Groups[1].Value, "", x.Loc });
                } else if (lang == "html") {
                    foreach (Match k in HtmlSrc.Matches(s)) { string v = k.Groups[1].Value; if (SkipHref(v)) continue; o.Add(new string[] { "src/href", v, "", x.Loc }); }
                    foreach (Match k in HtmlFn.Matches(s)) o.Add(new string[] { "static-file", k.Groups[1].Value, "", x.Loc });
                    foreach (Match k in JsImp.Matches(s)) o.Add(new string[] { "import", k.Groups[1].Value, "", x.Loc });
                } else if (lang == "css") {
                    foreach (Match k in CssImp.Matches(s)) { string v = k.Groups[1].Value; if (SkipHref(v)) continue; o.Add(new string[] { "css-url", v, "", x.Loc }); }
                } else if (lang == "bat" || lang == "ps" || lang == "vb") {
                    string tl = s.TrimStart();
                    if (lang == "bat" && (tl.StartsWith("rem ", StringComparison.OrdinalIgnoreCase) || tl.StartsWith("::"))) continue;
                    if (lang == "ps" && tl.StartsWith("#")) continue;
                    if (lang == "vb" && (tl.StartsWith("'") || tl.StartsWith("rem ", StringComparison.OrdinalIgnoreCase))) continue;
                    foreach (Match k in ExecTok.Matches(s)) o.Add(new string[] { "exec", k.Groups[1].Value, "", x.Loc });
                    if (lang == "ps") {
                        m = PsDot.Match(s); if (m.Success) o.Add(new string[] { "dot-source", m.Groups[1].Value, "", x.Loc });
                        m = PsMod.Match(s); if (m.Success) o.Add(new string[] { "module", m.Groups[1].Value, "", x.Loc });
                    }
                } else if (lang == "cs") {
                    m = CsUsing.Match(s); if (m.Success) o.Add(new string[] { "using", m.Groups[1].Value, "", x.Loc });
                } else if (lang == "java") {
                    m = JavaImp.Match(s); if (m.Success) o.Add(new string[] { "import", m.Groups[1].Value, "", x.Loc });
                } else if (lang == "rs") {
                    m = RsUse.Match(s); if (m.Success) o.Add(new string[] { m.Groups[1].Value, m.Groups[2].Value, "", x.Loc });
                }
            } catch (RegexMatchTimeoutException) { Tmo(x.Loc); }
        }
        return o;
    }

    static string NextName(List<Unit> u, int i, Regex r, int g) {
        for (int j = i + 1; j < u.Count && j <= i + 8; j++) {
            try { Match m = r.Match(u[j].Text); if (m.Success) return m.Groups[g].Value; } catch (RegexMatchTimeoutException) { Tmo(u[j].Loc); }

    public static List<string[]> Routes(List<Unit> u, string lang) {
        List<string[]> o = new List<string[]>();
        for (int i = 0; i < u.Count; i++) {
            try {
                string s = u[i].Text; Match m;
                if (lang == "py") {
                    m = PyRoute.Match(s);
                    if (m.Success) {
                        string meth = m.Groups[1].Value.ToUpperInvariant();
                        if (meth == "ROUTE") {
                            Match mm = PyMethods.Match(m.Groups[3].Value);
                            meth = mm.Success ? mm.Groups[1].Value.Replace("'", "").Replace("\"", "").Replace(" ", "").ToUpperInvariant() : "GET";
                        }
                        o.Add(new string[] { meth, m.Groups[2].Value, NextName(u, i, PyDef, 2), u[i].Loc });
                    }
                    Match ma = PyAddUrl.Match(s);
                    if (ma.Success) {
                        Match mm2 = PyMethods.Match(ma.Groups[3].Value);
                        string meth2 = mm2.Success ? mm2.Groups[1].Value.Replace("'", "").Replace("\"", "").Replace(" ", "").ToUpperInvariant() : "GET";
                        o.Add(new string[] { meth2, ma.Groups[2].Value, ma.Groups[4].Value, u[i].Loc });
                    }
                } else if (lang == "js") {
                    foreach (Match k in JsRoute.Matches(s)) o.Add(new string[] { k.Groups[1].Value.ToUpperInvariant(), k.Groups[2].Value, "", u[i].Loc });
                } else if (lang == "rs") {
                    m = RsRoute.Match(s);
                    if (m.Success) o.Add(new string[] { m.Groups[1].Value.ToUpperInvariant(), m.Groups[2].Value, NextName(u, i, RsFn, 1), u[i].Loc });
                    if (RsCmd.IsMatch(s)) { string fn = NextName(u, i, RsFn, 1); if (fn.Length > 0) o.Add(new string[] { "INVOKE", "invoke:" + fn, fn, u[i].Loc }); }
                }
            } catch (RegexMatchTimeoutException) { Tmo(u[i].Loc); }
        }
        return o;
    }

    // 0 = code, 1 = inside string literal, 2 = inside template interpolation ${ } (code)
    public static int QState(string s, int i, string qs) {
        char q = '\0'; bool ip = false; int dep = 0; char q2 = '\0';
        for (int j = 0; j < i && j < s.Length; j++) {
            char c = s[j];
            if (ip) {
                if (q2 != '\0') { if (c == '\\') { j++; continue; } if (c == q2) q2 = '\0'; continue; }
                if (c == '"' || c == '\'') { q2 = c; continue; }
                if (c == '{') dep++;
                else if (c == '}') { dep--; if (dep == 0) ip = false; }
                continue;
            }
            if (q != '\0') {
                if (c == '\\' && qs.Length > 1) { j++; continue; }
                if (q == '`' && c == '$' && j + 1 < s.Length && s[j + 1] == '{') { ip = true; dep = 1; j++; continue; }
                if (c == q) q = '\0';
                continue;
            }
            if (qs.IndexOf(c) >= 0) q = c;
        }
        if (ip) return q2 != '\0' ? 1 : 2;
        return q != '\0' ? 1 : 0;
    }

    // line-level context : comment / doc (python docstring) / block comment ; null = decide per token
    public class LineCtx { public bool InDoc; public bool InBlk; }
    public static string LineContext(string s, string lang, LineCtx st) {
        string t = s.TrimStart();
        if (lang == "") return "text";
        if (lang == "py") { if (st.InDoc) return "doc"; if (t.StartsWith("#")) return "comment"; }
        else if (lang == "ps") { if (t.StartsWith("#")) return "comment"; }
        else if (lang == "bat") { if (t.StartsWith("rem ", StringComparison.OrdinalIgnoreCase) || t.Equals("rem", StringComparison.OrdinalIgnoreCase) || t.StartsWith("::")) return "comment"; }
        else if (lang == "vb") { if (t.StartsWith("'") || t.StartsWith("rem ", StringComparison.OrdinalIgnoreCase)) return "comment"; }
        else if (lang == "sql") { if (t.StartsWith("--")) return "comment"; }
        else if (lang == "js" || lang == "cs" || lang == "java" || lang == "rs" || lang == "css" || lang == "html") { if (st.InBlk || t.StartsWith("//") || t.StartsWith("<!--")) return "comment"; }
        return null;
    }
    public static void UpdateLineCtx(string s, string lang, LineCtx st) {
        if (lang == "py") { int n = CountOf(s, "\"\"\"") + CountOf(s, "'''"); if (n % 2 == 1) st.InDoc = !st.InDoc; }
        if (lang == "js" || lang == "cs" || lang == "java" || lang == "rs" || lang == "css" || lang == "html") {
            int a = Math.Max(s.LastIndexOf("/*", StringComparison.Ordinal), s.LastIndexOf("<!--", StringComparison.Ordinal));
            int b = Math.Max(s.LastIndexOf("*/", StringComparison.Ordinal), s.LastIndexOf("-->", StringComparison.Ordinal));
            if (a >= 0 && a > b) st.InBlk = true; else if (b >= 0) st.InBlk = false;
        }
    }

    static int Braces(string s, int from, string qs, int dep, ref bool started) {
        char q = '\0';
        for (int j = from; j < s.Length; j++) {
            char c = s[j];
            if (q != '\0') { if (c == '\\') { j++; continue; } if (c == q) q = '\0'; continue; }
            if (qs.IndexOf(c) >= 0) { q = c; continue; }
            if (c == '{') { dep++; started = true; }
            else if (c == '}') { dep--; if (started && dep <= 0) return 0; }
        }
        return dep;
    }
    static string BodyText(List<Unit> u, int i, int at, string lang) {
        StringBuilder sb = new StringBuilder();
        if (lang == "py") {
            int ind = u[i].Text.Length - u[i].Text.TrimStart().Length;
            sb.AppendLine(u[i].Text);
            for (int j = i + 1; j < u.Count && j <= i + 400; j++) {
                string t = u[j].Text;
                if (t.Trim().Length == 0) { continue; }
                if (t.Length - t.TrimStart().Length <= ind) break;
                sb.AppendLine(t);
                if (sb.Length > 20000) break;
            }
            return sb.ToString();
        }
        string qs = QuoteChars(lang);
        string s0 = u[i].Text;
        int ar = s0.IndexOf("=>", at, StringComparison.Ordinal);
        if (ar >= 0 && !s0.Substring(ar + 2).TrimStart().StartsWith("{")) return s0;
        bool started = false;
        int dep = Braces(s0, at, qs, 0, ref started);
        sb.AppendLine(s0);
        if (started && dep <= 0) return sb.ToString();
        for (int j = i + 1; j < u.Count && j <= i + 400; j++) {
            sb.AppendLine(u[j].Text);
            dep = Braces(u[j].Text, 0, qs, dep, ref started);
            if (started && dep <= 0) break;
            if (sb.Length > 20000) break;
        }
        return sb.ToString();
    }
    // wrapper functions defined in a file (all functions are listed; Http=true only if the body calls http)
    public static List<Wrap> Wrappers(List<Unit> u, string lang, string file) {
        List<Wrap> o = new List<Wrap>();
        if (lang != "js" && lang != "html" && lang != "py") return o;
        for (int i = 0; i < u.Count; i++) {
            try {
                string s = u[i].Text;
                string name = null, prm = null; int at = 0;
                if (lang == "py") {
                    Match m = PyDefP.Match(s);
                    if (!m.Success) continue;
                    name = m.Groups[2].Value; prm = m.Groups[3].Value; at = m.Index;
                } else {
                    Match m = JsDefP.Match(s);
                    if (!m.Success) continue;
                    if (m.Groups[1].Success) { name = m.Groups[1].Value; prm = m.Groups[2].Value; }
                    else { name = m.Groups[3].Value; prm = m.Groups[4].Success ? m.Groups[4].Value : m.Groups[5].Success ? m.Groups[5].Value : m.Groups[6].Value; }
                    at = m.Index;
                }
                Wrap w = new Wrap(); w.Name = name; w.File = file; w.Lang = lang; w.Loc = u[i].Loc;
                string[] ps = prm.Split(',');
                int idx = 0;
                for (int k = 0; k < ps.Length; k++) {
                    string p = ps[k].Trim();
                    if (lang == "py" && k == 0 && (p == "self" || p == "cls")) continue;
                    Match mp = MethParam.Match(p);
                    if (mp.Success && p.Length > 0) { w.MIdx = idx; if (mp.Groups[1].Success) { w.Def = mp.Groups[1].Value.ToUpperInvariant(); w.DefBasis = "parameter-default"; } }
                    idx++;
                }
                w.Body = BodyText(u, i, at, lang);
                w.Http = HttpPrim.IsMatch(w.Body);
                if (w.Http) { w.Via = "primitive"; w.Depth = 0; }
                if (w.Def == null) {
                    MatchCollection mbs = MethBody.Matches(w.Body);
                    HashSet<string> bodyMethods = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
                    foreach (Match bx in mbs) bodyMethods.Add(bx.Groups[1].Value.ToUpperInvariant());
                    MatchCollection pvs = PrimVerb.Matches(w.Body);
                    foreach (Match px in pvs) bodyMethods.Add(px.Groups[1].Value.ToUpperInvariant());
                    // A body-level method is a safe default only when every observed primitive agrees.
                    if (bodyMethods.Count == 1) { foreach (string mv in bodyMethods) w.Def = mv; w.DefBasis = "single-body-method"; }
                    else if (bodyMethods.Count > 1) { w.Def = null; w.DefBasis = "multiple-body-methods"; }
                }
                o.Add(w);
            } catch (RegexMatchTimeoutException) { Tmo(u[i].Loc); }
        }
        return o;
    }
    // wrapper calling wrapper : http flag and default method are inherited (fixed point)
    // v5.10 : breadth-first inheritance, limited to maxDepth call levels.
    // depth 0 = calls fetch / axios / requests ... directly, depth N = calls a depth N-1 wrapper.
    public static void FinalizeWraps(Dictionary<string, Wrap> d, int maxDepth) {
        for (int depth = 1; depth <= maxDepth; depth++) {
            List<Wrap> newly = new List<Wrap>();
            foreach (Wrap w in d.Values) {
                if (w.Http || w.Body == null) continue;
                foreach (Wrap w2 in d.Values) {
                    if (w2 == w || !w2.Http || w2.Depth != depth - 1 || w2.Lang != w.Lang) continue;
                    bool hit = false;
                    try { hit = Regex.IsMatch(w.Body, @"(?<![\w$.])" + Regex.Escape(w2.Name) + @"\s*\(", RegexOptions.None, TimeSpan.FromSeconds(1)); } catch { continue; }
                    if (!hit) continue;
                    newly.Add(w); w.Via = "calls " + w2.Name;
                    // Inherited methods are hints only. Route matching may later reject this hint.
                    if (w.Def == null && w.MIdx < 0 && w2.Def != null && w2.MIdx < 0 && w2.DefBasis != "multiple-body-methods") { w.Def = w2.Def; w.DefBasis = "inherited:" + w2.Name; }
                    break;
                }
            }
            if (newly.Count == 0) break;
            foreach (Wrap w in newly) { w.Http = true; w.Depth = depth; }
        }
        foreach (Wrap w in d.Values) {
            if (w.Http && w.Def == null && w.DefBasis != "multiple-body-methods" && (w.Body.Contains("fetch(") || w.Body.Contains("fetch (") || w.Body.Contains("urlopen("))) { w.Def = "GET"; w.DefBasis = "primitive-default"; }
        }
    }
    static void FindCallee(string s, int idx, out string callee, out int open) {
        callee = ""; open = -1; int depth = 0;
        for (int j = idx - 1; j >= 0; j--) {
            char c = s[j];
            if (c == ')' || c == ']' || c == '}') depth++;
            else if (c == '(' || c == '[' || c == '{') {
                if (depth > 0) { depth--; continue; }
                if (c != '(') continue;
                int k = j - 1;
                while (k >= 0 && s[k] == ' ') k--;
                int e = k;
                while (k >= 0 && (char.IsLetterOrDigit(s[k]) || s[k] == '_' || s[k] == '$' || s[k] == '.')) k--;
                string nm = e >= 0 ? s.Substring(k + 1, e - k) : "";
                if (nm.Length == 0) continue;
                if (Kw.Contains(nm)) return;
                callee = nm; open = j; return;
            }
            else if (c == ';' && depth == 0) return;
        }
    }
    static List<string> SplitArgs(string s, int open, int urlPos, out int urlArg) {
        List<string> a = new List<string>(); urlArg = -1;
        int depth = 0; char q = '\0'; int st = open + 1;
        int j = open + 1;
        for (; j < s.Length; j++) {
            char c = s[j];
            if (q != '\0') { if (c == '\\') { j++; continue; } if (c == q) q = '\0'; continue; }
            if (c == '"' || c == '\'' || c == '`') { q = c; continue; }
            if (c == '(' || c == '[' || c == '{') depth++;
            else if (c == ')' || c == ']' || c == '}') { if (depth == 0) break; depth--; }
            else if (c == ',' && depth == 0) { if (urlPos >= st && urlPos < j) urlArg = a.Count; a.Add(s.Substring(st, j - st)); st = j + 1; }
        }
        int en = Math.Min(j, s.Length);
        if (en > st || a.Count > 0) { if (urlPos >= st && urlPos < en) urlArg = a.Count; a.Add(s.Substring(st, Math.Max(0, en - st))); }
        return a;
    }
    // raw call sites (method resolved later with wrappers of all files)
    // row : path, dynamic, callee, method, basis, argLiterals(\u001f), urlArg, loc
    public static List<string[]> ClientRaw(List<Unit> u, string lang) {
        List<string[]> o = new List<string[]>();
        if (lang != "js" && lang != "html" && lang != "py") return o;
        string qs = QuoteChars(lang);
        foreach (Unit x in u) {
            string s = x.Text;
            try {
                // v5.2 generic separation of server declarations from client calls.
                // A server route declaration describes an endpoint; it is not an outgoing HTTP request.
                // PyRoute covers decorator-style Python servers (Flask/FastAPI-like).
                // JsRoute covers app/router/server .get/.post/... declarations.
                if ((lang == "py" && PyRoute.IsMatch(s)) || (lang == "js" && JsRoute.IsMatch(s))) continue;
                foreach (Match m in UrlLit2.Matches(s)) {
                    if (QState(s, m.Index, qs) == 1) continue;
                    string path = m.Groups[2].Value;
                    if (path.StartsWith("http", StringComparison.OrdinalIgnoreCase)) {
                        int p = path.IndexOf("//"); int q = p >= 0 ? path.IndexOf('/', p + 2) : -1;
                        if (q < 0) continue;
                        path = path.Substring(q);
                    }
                    bool dyn = path.Contains("${");
                    path = Regex.Replace(path, @"\$\{[^}]*\}", "{}");
                    string after = s.Substring(m.Index + m.Length);
                    if (path.EndsWith("/") && Regex.IsMatch(after, @"^\s*\+")) { path += "{}"; dyn = true; }
                    string callee; int open;
                    FindCallee(s, m.Index, out callee, out open);
                    string meth = "", basis = "unknown"; string lits = ""; int ua = -1;
                    if (open >= 0) {
                        List<string> args = SplitArgs(s, open, m.Index, out ua);
                        List<string> lv = new List<string>();
                        bool ajax = callee.IndexOf("ajax", StringComparison.OrdinalIgnoreCase) >= 0;
                        foreach (string a in args) {
                            Match la = LitArg.Match(a); lv.Add(la.Success ? la.Groups[1].Value : "");
                            if (meth.Length == 0) {
                                Match ma = MethArg.Match(a);
                                if (!ma.Success && ajax) ma = TypeArg.Match(a);
                                if (ma.Success) { meth = ma.Groups[1].Value.ToUpperInvariant(); basis = "option"; }
                            }
                        }
                        lits = string.Join("\u001f", lv.ToArray());
                        string last = callee.Substring(callee.LastIndexOf('.') + 1);
                        if (meth.Length == 0 && callee.Contains(".") && Verbs.Contains(last)) { meth = last.ToUpperInvariant(); basis = "callee-verb"; }
                        if (meth.Length == 0 && last == "fetch") { meth = "GET"; basis = "fetch-default"; }
                    }
                    o.Add(new string[] { path, dyn ? "yes" : "no", callee, meth, basis, lits, ua.ToString(), x.Loc });
                }
                foreach (Match k in Invoke.Matches(s)) o.Add(new string[] { "invoke:" + k.Groups[1].Value, "no", "invoke", "INVOKE", "tauri-invoke", "", "0", x.Loc });
            } catch (RegexMatchTimeoutException) { Tmo(x.Loc); }
        }
        return o;
    }

    static Dictionary<string, Regex> _rc = new Dictionary<string, Regex>();
    public static bool RouteMatch(string route, string url) {
        if (string.IsNullOrEmpty(route) || string.IsNullOrEmpty(url)) return false;
        if (route.StartsWith("invoke:") || url.StartsWith("invoke:")) return route == url;
        Regex r;
        if (!_rc.TryGetValue(route, out r)) {
            string t = Regex.Replace(route, @"<[^>]+>|\{[^}]+\}|(?<=/):[A-Za-z_]\w*|\*", "\u0001");
            string p = Regex.Escape(t).Replace("\u0001", "[^/]+");
            p = "^" + p.TrimEnd('/') + "/?$";
            r = new Regex(p, RegexOptions.CultureInvariant);
            _rc[route] = r;
        }
        return r.IsMatch(url.Replace("{}", "x1"));
    }
    // v5.4 : generic route specificity. Higher is more specific.
    // Static/exact routes outrank parameterized routes; among dynamic routes, fewer
    // parameters and more literal characters win. No framework/app-specific names.
    public static int RouteSpecificity(string route, string url) {
        if (string.IsNullOrEmpty(route) || string.IsNullOrEmpty(url)) return -1;
        if (!RouteMatch(route, url)) return -1;
        if (route.StartsWith("invoke:") || url.StartsWith("invoke:")) return route == url ? 1000000 : -1;
        string rn = route.TrimEnd('/'); string un = url.TrimEnd('/');
        bool dyn = Regex.IsMatch(rn, @"<[^>]+>|\{[^}]+\}|(?<=/):[A-Za-z_]\w*|\*");
        if (!dyn && string.Equals(rn, un, StringComparison.Ordinal)) return 1000000 + rn.Length;
        MatchCollection mm = Regex.Matches(rn, @"<[^>]+>|\{[^}]+\}|(?<=/):[A-Za-z_]\w*|\*");
        string lit = Regex.Replace(rn, @"<[^>]+>|\{[^}]+\}|(?<=/):[A-Za-z_]\w*|\*", "");
        int seg = rn.Split(new char[]{'/'}, StringSplitOptions.RemoveEmptyEntries).Length;
        return 100000 + lit.Length * 100 + seg * 10 - mm.Count * 10000;
    }

    static readonly Regex WriteIoRx = RI(@"\b(?:write_text|write_bytes|writeFile\w*|createWriteStream|Set-Content|Add-Content|Out-File|Export-Csv|to_csv|to_excel|dump|dumps|save|saveas|touch|mkdir|makedirs|replace|rename|Copy-Item|Move-Item)\s*\(|\bopen\s*\([^\r\n]*['""](?:w|a|x|wb|ab|xb)['""]|(?:^|\s)(?:copy|move|xcopy|robocopy)\s");

    static readonly string[] CtxOrder = new string[] { "write-io", "io", "quoted", "text", "plain", "doc", "comment", "attr" };

                        else c = "plain";
                    }
                    Ag a = Bump(d, order, v.ToLowerInvariant(), v, x.Loc);
                    int n; a.C.TryGetValue(c, out n); a.C[c] = n + 1;
                }
            } catch (RegexMatchTimeoutException) { Tmo(x.Loc); }

        }

            o.Add(new string[] { a.K, a.N.ToString(), a.Loc, string.Join(" ", sm.ToArray()), cs.Count > 0 ? cs[0] : "-" });
        }
        return o;
    }

function SlowDump { param([string]$src, [long]$limit)
    $hx = @(0..255 | ForEach-Object { $_.ToString('X2') })
    $reportWriterLocal = $REPORT_WRITER
    $fs = [IO.File]::Open($src, 'Open', 'Read', 'ReadWrite')
    $off = [long]0
    try {
        $reportWriterLocal.WriteLine('[HEX DUMP]')
        $reportWriterLocal.WriteLine('Offset    00 01 02 03 04 05 06 07  08 09 0A 0B 0C 0D 0E 0F  ASCII')
        $buf = New-Object byte[] 16
        while ($true) {
            $r = $fs.Read($buf, 0, 16)
            if ($r -le 0) { break }
            $sb = New-Object System.Text.StringBuilder
            [void]$sb.Append($off.ToString('X8')).Append('  ')
            for ($j = 0; $j -lt 16; $j++) {
                if ($j -lt $r) { [void]$sb.Append($hx[$buf[$j]]).Append(' ') } else { [void]$sb.Append('   ') }
                if ($j -eq 7) { [void]$sb.Append(' ') }
            }
            [void]$sb.Append(' ')
            for ($j = 0; $j -lt $r; $j++) { $v = $buf[$j]; if (($v -ge 32) -and ($v -lt 127)) { [void]$sb.Append([char]$v) } else { [void]$sb.Append('.') } }
            $reportWriterLocal.WriteLine($sb.ToString())
            $off += $r
            if (($limit -gt 0) -and ($off -ge $limit)) { break }
        }
        $reportWriterLocal.WriteLine('[END: ' + $off + ' bytes dumped]')
    } finally { $fs.Close() }
    return $off
}

        }

        # [16-3] client calls
        L '---- [16-3] CLIENT CALLS (url literal + HTTP method) -> matched server route ----'
        L ' Method basis : option (method:/type: in call args) / callee-verb (x.get / x.post ...) /'
        L '                wrapper-arg(f) (method argument of wrapper f) / wrapper-default(f) / fetch-default / unknown'
        L ' Method strength: explicit / inferred / default-hint / rejected-default-hint / unknown'
        L ' A conflicting default-hint is kept in MethodHint but is not forced onto the route.'
        L ' Url "{}"     : dynamic segment ("/a/" + id, template ${..}) -> matches only routes with a parameter there'
        L ' Evidence     : match = selected route after method + specificity ranking'
        L '                candidate = method differs or multiple routes remain equally specific'
        L ' "(no route found)" = static resource, external api, or route not in the scanned folder.'
        L ' Specificity  : exact/static route > fewer parameters > more literal text; all alternatives are kept in http_route_candidates.'
        $rows = New-Object System.Collections.ArrayList
        $routeAudit = New-Object System.Collections.ArrayList
        foreach ($x in $fx) {
            $sm = $sem[[int]$x.No]
            if (-not $sm -or -not $sm.Raw) { continue }
            $calls = [SemTool]::ResolveCalls($sm.Raw, $wraps, [string]$sm.Lang)
            foreach ($r in $calls) {

                $methodStrength = 'unknown'
                if ($bas -in @('option','callee-verb','wrapper-arg')) { $methodStrength = 'explicit' }
                elseif ($bas -like 'wrapper-arg(*') { $methodStrength = 'explicit' }
                elseif (($bas -like 'wrapper-default(*') -or ($bas -eq 'fetch-default')) { $methodStrength = 'default-hint' }
                elseif ($meth) { $methodStrength = 'inferred' }
                $mshow = $meth; if (-not $mshow) { $mshow = '?' }

                if ($hitP.Count -eq 0) {

                    continue

                # none agree, keep all path matches as candidate (method differs).
                $pool = $hitP; $ev = 'match'; $note = ''

                        # A default from fetch/wrapper is not proof. Preserve it as MethodHint,
                        # but let path specificity select the route instead of forcing a wrong method.
                        $meth = ''; $mshow = '?'; $methodStrength = 'rejected-default-hint'
                        $note = ' (default hint conflicts; route-selected)'
                    } else {
                        $ev = 'candidate'; $note = ' (method differs)'
                    }
                }
                # Rank generic route specificity. Exact/static route wins over /<param>, /{param}, :param, *.
                $scored = @($pool | ForEach-Object { [pscustomobject]@{ H = $_; Score = [SemTool]::RouteSpecificity($_.Path, $url) } })
                $bestScore = ($scored | Measure-Object Score -Maximum).Maximum
                $best = @($scored | Where-Object { $_.Score -eq $bestScore })
                $ambig = ($best.Count -gt 1)
                if ($ambig) { $ev = 'candidate'; if (-not $note) { $note = ' (equal specificity)' } }
                # Preserve every matched alternative in audit TSV even when specificity suppresses it.
                foreach ($z in $scored) {
                    $sel = 'no'; if ($z.Score -eq $bestScore) { $sel = 'yes' }
                    [void]$routeAudit.Add([string[]]@([string]$x.No, (FP $x.No), $mshow, $methodHint, $methodStrength, $url, [string]$z.Score, $sel, [string]$z.H.No, (FP $z.H.No), $z.H.Method, $z.H.Path, $z.H.Handler, $bas))
                }
                foreach ($z in $best) {
                    $h = $z.H
                    [void]$rows.Add([string[]]@([string]$x.No, (FP $x.No), $url, $r[5], $r[6], ($h.Method + ' ' + $h.Path + $note), $h.Handler, [string]$h.No, (FP $h.No), $mshow, $bas, $methodStrength, $methodHint, $cal, $dyn, $ev, [string]$z.Score, ($scored.Count - 1)))
                    AddE 'HTTP' $x.No $h.No ($mshow + ' ' + $url + ' => ' + $h.Method + ' ' + $h.Path + ' (' + $h.Handler + ')' + $note + ' [' + $bas + '; method-strength=' + $methodStrength + '; method-hint=' + $methodHint + '; specificity=' + $z.Score + '; alternatives=' + ($scored.Count - 1) + ']') $r[6] $ev
                }
            }
        }
        if ($rows.Count -eq 0) { L '(no client url found)' }
        TSV 'client_calls' @('No', 'Path', 'Url', 'Hits', 'FirstLoc', 'MatchedRoute', 'Handler', 'ServerNo', 'ServerPath', 'Method', 'MethodBasis', 'MethodStrength', 'MethodHint', 'Callee', 'Dynamic', 'Evidence', 'Specificity', 'SuppressedAlternatives') $rows

        if ($routeAudit.Count -eq 0) { L '(none)' }
        TSV 'http_route_candidates' @('ClientNo','ClientPath','EffectiveMethod','MethodHint','MethodStrength','Url','Specificity','Selected','ServerNo','ServerPath','RouteMethod','Route','Handler','MethodBasis') $routeAudit

        # [16-4] config keys
        L '---- [16-4] CONFIG / JSON KEY USE ----'
        L ' UsedByNo  = source files that contain the key as a quoted string (same as v4, kept)'
        L ' AccessNo  = source files that access the key : x["key"] / .get("key") / getattr(x, "key") ...'
        L ' Generic   = key name in GENERICKEYS (common word, low evidence)'
        L ' Edge evidence per (code file, json file) :'
        L '   match     = code file writes the json file name ([11]) and uses its keys'
        L '   candidate = no file name, but specific keys are accessed (1+) or written (2+)'
        L '   weak      = only generic keys or a single specific literal (kept, probably coincidence)'
        $rows = New-Object System.Collections.ArrayList
        $codeX = @($fx | Where-Object { $s0 = $sem[[int]$_.No]; $s0 -and $s0.Lit })
        foreach ($x in $fx) {
            $sm = $sem[[int]$x.No]
            if (-not $sm -or -not $sm.JK) { continue }
            $byUser = @{}
            foreach ($r in $sm.JK) {
                $key = $r[0]
                $gen = 'no'; if ($GENERICKEYS -contains $key.ToLowerInvariant()) { $gen = 'yes' }

                        $bu = $byUser[[int]$c.No]
                        [void]$bu.Keys.Add($key)
                        if ($gen -eq 'no') { if ($isA) { $bu.SpecA++ } elseif ($isL) { $bu.SpecL++ } }
                    }
                }
                $ub = '(not found in code)'; if ($users.Count -gt 0) { $ub = $users -join ',' }
                $ab = '-'; if ($accs.Count -gt 0) { $ab = $accs -join ',' }
                [void]$rows.Add([string[]]@([string]$x.No, (FP $x.No), $key, $r[1], $r[2], $ub, $ab, $gen))
            }
            foreach ($kk in $byUser.Keys) {
                $bu = $byUser[$kk]
                $ks = @($bu.Keys)
                $ex = ($ks | Select-Object -First 8) -join ' '
                if ($refPair.ContainsKey(([string]$kk + '|' + [string]$x.No))) { $ev = 'match'; $why = 'file name written' }
                elseif (($bu.SpecA -ge 1) -or ($bu.SpecL -ge 2)) { $ev = 'candidate'; $why = 'specific keys' }
                else { $ev = 'weak'; $why = 'generic or single key only' }
                AddE 'CONFIG' $kk $x.No ('keys=' + $ks.Count + ' specific-access=' + $bu.SpecA + ' specific-literal=' + $bu.SpecL + ' (' + $why + ') e.g. ' + $ex) '' $ev
            }

        TSV 'config_keys' @('JsonNo', 'JsonPath', 'Key', 'Count', 'FirstLoc', 'UsedByNo', 'AccessNo', 'Generic') $rows

        # [16-5] sql

        # [16-5a/b] database discovery / linkage (v5.18)
        L '---- [16-5a] DATABASE FILES / PAIRS ----'

        function DbRole { param($q)
            $e0 = ([string]$q.File.Extension).ToLowerInvariant()
            if ($e0 -eq '.mdf') { return 'sqlserver-data' }
            if ($e0 -eq '.ldf') { return 'sqlserver-log' }
            if ($e0 -in @('.sqlite','.sqlite3')) { return 'sqlite' }
            if ($e0 -in @('.mdb','.accdb')) { return 'access' }
            if ($e0 -eq '.db') { return 'generic-db' }
            return ''
        }
        function DbKey { param($q)
            $bn0 = [string]$q.File.BaseName
            if ($q.File.Extension -ieq '.ldf') { $bn0 = $bn0 -replace '(?i)_log$','' }
            return (($q.File.DirectoryName.TrimEnd('\') + '\' + $bn0).ToLowerInvariant())
        }

        $dbMap = @{}
        foreach ($d0 in $dbX) { $dk0 = DbKey $d0; if (-not $dbMap.ContainsKey($dk0)) { $dbMap[$dk0] = New-Object System.Collections.ArrayList }; [void]$dbMap[$dk0].Add($d0) }
        $dbRows = New-Object System.Collections.ArrayList

                if ($mate0.Count -gt 0) {
                    $pairNo = [string]$mate0[0].No; $pairPath = FP $mate0[0].No; $status0 = 'paired'; $ev0 = 'same-folder-base+extension'
                    $pk0 = @([int]$d0.No,[int]$mate0[0].No) | Sort-Object; $pks0 = ($pk0 -join '|')
                    if (-not $dbPairSeen.ContainsKey($pks0)) { AddE 'DBFILE' $d0.No $mate0[0].No ('database-pair ' + $role0 + ' <-> ' + (DbRole $mate0[0])) '' 'fact'; $dbPairSeen[$pks0] = 1 }
                } else { $status0 = 'pair-missing' }

        }

                    }

        L
        # [16-6] file names with context
        L '---- [16-6] FILE NAMES WRITTEN IN FILES (incl. files NOT in the scanned folder) ----'
        L ' Status  : internal(see [11]) = exists in folder / pattern-matched = wildcard name matched files'
        L '           external = not in the scanned folder (input data, missing file, runtime output ...)'
        L '           external-pattern = wildcard name with no match in folder / self = own file name'

        TSV 'file_tokens' @('No', 'Path', 'Token', 'Hits', 'FirstLoc', 'Basis', 'Status', 'MatchedNo', 'Context', 'BestContext') $rows
        L
        # [16-7] symbol use across files
        L ('---- [16-7] SYMBOL USE ACROSS FILES (name defined in one file appears in another file of the same language family, name length >= ' + $SYMMIN + ') ----')
        L ' Candidate : same name may be a different thing. Combine with [16-1] import to confirm.'
        $fam = @{ py = 'py'; js = 'js'; html = 'js'; cs = 'net'; vb = 'net'; java = 'java'; rs = 'rs'; ps = 'ps'; bat = 'bat'; sql = 'sql' }

                  'status', 'start', 'stop', 'test', 'list', 'dict', 'text', 'file', 'files', 'type', 'info', 'error', 'result', 'args',
                  'kwargs', 'true', 'false', 'none', 'null', 'return', 'function', 'const', 'async', 'await', 'class', 'import', 'from',
                  'index', 'count', 'item', 'node', 'size', 'time', 'date', 'line', 'temp', 'json', 'html', 'string', 'object')
        $skipKind = @('label', 'var', 'top-var', 'method/inner-def')

        foreach ($x in $fx) {

            foreach ($r in $sm.Sym) {
                if ($skipKind -contains $r[0]) { continue }

                if (-not $defs.ContainsKey($dk)) { $defs[$dk] = New-Object System.Collections.ArrayList }

            }

            $f0 = $fam[$sm.Lang]
            $use = @{}
            foreach ($dk in $defs.Keys) {
                $pp = $dk.Split('|')
                if ($pp[0] -ne $f0) { continue }
                if ($defs[$dk].Contains([string]$x.No)) { continue }
                if (-not $sm.Idn.Contains($pp[1])) { continue }
                foreach ($d in $defs[$dk]) {
                    if (-not $use.ContainsKey($d)) { $use[$d] = New-Object System.Collections.ArrayList }
                    [void]$use[$d].Add($pp[1])
                }
            }
            foreach ($d in $use.Keys) {
                $ns = @($use[$d] | Sort-Object)
                $ex = ($ns | Select-Object -First 10) -join ' '
                [void]$rows.Add([string[]]@([string]$x.No, (FP $x.No), [string]$d, (FP $d), [string]$ns.Count, ($ns -join ' ')))
                AddE 'SYM' $x.No $d ('names=' + $ns.Count + ' e.g. ' + $ex) '' 'candidate'
            }
        }
        if ($rows.Count -eq 0) { L '(no cross-file symbol use found)' }
        TSV 'symbol_use' @('UserNo', 'UserPath', 'DefNo', 'DefPath', 'Names', 'NameList') $rows
        L
        # ---------- [17] integrated relation map ----------
        Write-Host '[17/19] Integrated relation map...'
        foreach ($r in $refs) { foreach ($t in $r[2].Split(',')) { AddE 'REF' $r[0] $t ('token=' + $r[1] + ' hits=' + $r[3]) $r[4] 'fact' } }
        foreach ($p in $pairs) { AddE 'KEY' $p[0] $p[1] ('shared=' + $p[2] + ' e.g. ' + $p[3]) '' 'candidate' }
        foreach ($g in $gs) {
            $gl = @($g.Group)
            for ($i = 0; $i -lt $gl.Count; $i++) { for ($j = $i + 1; $j -lt $gl.Count; $j++) { AddE 'SAME' $gl[$i].No $gl[$j].No ('sha256=' + $g.Name.Substring(0, 16) + '..') '' 'fact' } }

        function ExtClass { param($e)
            $ty = [string]$e[0]; $d = [string]$e[3]

                if ($ctx -eq 'attr') { return 'lexical-only' }
                $tok = [regex]::Replace($d, '\s*\[[a-z\-]+\]\s*$', ''); $isPattern = $tok.StartsWith('pattern ') -or ($tok -match '[*?]'); $tok = $tok -replace '^pattern ', ''
                $leaf = ([SemTool]::BaseName($tok)).ToLowerInvariant(); $sx = $fx[[int]$e[1] - 1]; $sl = [string]$sx.Lang
                if ($isPattern) { return 'file-pattern' }
                if ($leaf -match '\.(exe|com|dll|ocx|sys|drv|jar|bat|cmd|ps1|vbs)$') { return 'external-program' }
                if (($tok -match '(^|[\\/])(xl|word|ppt|docProps|_rels|customXml)[\\/]') -or ($leaf -eq '[content_types].xml')) { return 'archive-internal-part' }
                if ($ctx -eq 'write-io') { return 'output-file-candidate' }
                if (($ctx -eq 'io') -and $writtenLeaves.ContainsKey($leaf)) { return 'runtime-work-file' }
                if (($leaf -match '\.(log|tmp|temp|bak|pid|lock|cache)$') -or ($leaf -match '\.lock\.')) { return 'runtime-file-candidate' }
                if ($sl -eq 'json') { return 'configured-data-path' }
                if ($sl -eq 'html') { return 'ui-example-or-reference' }
                if ($e[6] -eq 'binary-string') { return 'binary-embedded-name' }
                if ($e[6] -eq 'documentation') { return 'documentation-mention' }
                if ($ctx -eq 'text') { return 'text-mention' }
                if ($ctx -eq 'io') { return 'external-file' }
                if ($ctx -in @('quoted','plain')) { return 'external-data-reference' }
                return 'external-file'
            }
            if ($ty -eq 'IMPORT') {
                $pp0 = $d.Split(' '); $mod = ''; if ($pp0.Count -ge 2) { $mod = $pp0[1] }
                $lg = [string]$fx[[int]$e[1] - 1].Lang
                if ($lg -eq 'py') {
                    if ($mod.StartsWith('.')) { return 'local-module-missing' }
                    $root = $mod.Split('.')[0].ToLowerInvariant()
                    if ($PYSTD -contains $root) { return 'stdlib' }
                    if ($declDeps.ContainsKey(($root -replace '[-.]', '_'))) { return 'third-party-declared' }
                    if ($PYALIAS.ContainsKey($root)) { $pk0 = ($PYALIAS[$root] -replace '[-.]', '_'); if ($declDeps.ContainsKey($pk0)) { return 'third-party-declared' }; if ($root -in @('werkzeug','jinja2') -and $declDeps.ContainsKey('flask')) { return 'third-party-declared' }; return 'third-party-known-undeclared' }
                    if (-not $hasManifest) { return 'third-party-no-manifest' }
                    return 'third-party-or-local-undeclared'
                }
                if ($lg -in @('js', 'html')) {
                    if ($mod -match '^(\.|/)') { return 'local-module-missing' }
                    $m2 = $mod -replace '^node:', ''
                    $root = $m2.Split('/')[0].ToLowerInvariant()
                    if ($m2.StartsWith('@')) { $sp = $m2.Split('/'); if ($sp.Count -ge 2) { $root = ($sp[0] + '/' + $sp[1]).ToLowerInvariant() } }
                    if (($NODESTD -contains $root) -or ($NODESTD -contains $m2.ToLowerInvariant())) { return 'stdlib' }
                    if ($declDeps.ContainsKey(($root -replace '[-.]', '_'))) { return 'third-party-declared' }
                    return 'third-party-or-local-undeclared'
                }
                if ($lg -in @('cs', 'java', 'rs', 'ps', 'vb')) { return 'platform-or-library' }
                return 'external-module'
            }
            if ($ty -eq 'INCLUDE') { return 'missing-asset-or-served-elsewhere' }
            if ($ty -eq 'EXEC') { return 'external-program' }
            if ($ty -eq 'SQL') { if ($e[5] -eq 'weak') { return 'sql-alias-like' }; return 'external-table' }
            return 'external-other'

        function MentionRule { param($gx)
            $full = [regex]::Escape([string]$fx[[int]$gx.No - 1].File.Name)
            $all = @($gx.Out | Where-Object { [string]$_ -match '^\s+mention (?!:)' })
            $real = @($all | Where-Object { [string]$_ -notmatch ' L\d+:\s*(from|import)\s' })
            $hf = @($real | Where-Object { [string]$_ -match ('(?i)' + $full) })
            if ($hf.Count -gt 0) { Explain $gx 'mentioned-in-other-file' ([string]$hf[0]) }
            elseif ($real.Count -gt 0) { Explain $gx 'basename-mentioned' ([string]$real[0]) }
        }
        function TextHit { param([string]$needle, $skipNo, $origins, [string]$exclude = '')
            if (-not $needle) { return '' }
            foreach ($q0 in $fx) {
                if (($q0.Kind -ne 'text') -or ($q0.No -eq $skipNo)) { continue }
                if ($origins -and ((EdgeOrigin $q0.No) -notin $origins)) { continue }
                $arr = GapLines $q0.No
                for ($i0 = 0; $i0 -lt $arr.Count; $i0++) {
                    if ($arr[$i0].IndexOf($needle, [StringComparison]::OrdinalIgnoreCase) -lt 0) { continue }
                    if ($exclude -and ($arr[$i0].IndexOf($exclude, [StringComparison]::OrdinalIgnoreCase) -ge 0)) { continue }
                    return ((FP $q0.No) + ' L' + ($i0 + 1) + ': ' + (GClip $arr[$i0]))
                }
            }
            return ''
        }
        function FileRules { param($gx)
            if ($gx.Kind -notin @('isolated-file', 'role-unclassified')) { return }
            if (-not $gx.No) { return }
            $q = $fx[[int]$gx.No - 1]; $nm = [string]$q.File.Name; $dir = [string]$q.File.DirectoryName; $bn = [string]$q.File.BaseName; $ex = [string]$q.File.Extension
            # template-of
            $mt = [regex]::Match($nm, '^(.+?)\.(sample|example|template|dist|default|orig)(\.[A-Za-z0-9]+)?$', 'IgnoreCase')
            if ($mt.Success) {
                $tgt = $mt.Groups[1].Value + $mt.Groups[3].Value
                $h = TextHit $tgt $q.No $null $nm
                if ($h) { Explain $gx 'template-of' ('template of ' + $tgt + ' ; ' + $h) 'strong'; return }
            }
            # superseded-by-sibling
            foreach ($sb in @($fx | Where-Object { ($_.No -ne $q.No) -and ([string]$_.File.DirectoryName -eq $dir) -and ($_.File.BaseName -ieq $bn) })) {
                $h = TextHit ([string]$sb.File.Name) $q.No $null $nm
                if ($h) {
                    $ev = 'sibling ' + $sb.File.Name + ' is referenced : ' + $h
                    $d0 = TextHit ('*' + $ex) $q.No @('documentation', 'text-data')
                    if ($d0) { $ev = $ev + ' | ' + $d0 }
                    Explain $gx 'superseded-by-sibling' $ev 'strong'; return
                }
            }
            # naming-convention
            $lit = TextHit ('"' + $bn + '"') $q.No @('source-code')
            if (-not $lit) { $lit = TextHit ("'" + $bn + "'") $q.No @('source-code') }
            if ($lit -and $ex) {
                foreach ($pr in @($fx | Where-Object { ($_.No -ne $q.No) -and ([string]$_.File.DirectoryName -eq $dir) -and ($_.File.Extension -ieq $ex) })) {
                    $h = TextHit ([string]$pr.File.Name) $q.No $null
                    if ($h) { Explain $gx 'naming-convention' ('key "' + $bn + '" in code : ' + $lit + ' | peer ' + $pr.File.Name + ' (same folder, same extension) referenced : ' + $h) 'strong'; return }
                }
            }
            # folder-description
            if (($q.Kind -eq 'text') -and ($ex.ToLowerInvariant() -in @('.txt', '.md', '.rst'))) {
                $txt = (GapLines $q.No) -join "`n"
                $cnt = 0; $exs = @()
                foreach ($k in $names.Keys) {
                    if (($k.Length -lt 5) -or ($k -eq $nm.ToLowerInvariant())) { continue }
                    if ($txt.IndexOf($k, [StringComparison]::OrdinalIgnoreCase) -ge 0) { $cnt++; if ($exs.Count -lt 6) { $exs += $k } }
                }
                if ($cnt -ge 5) { Explain $gx 'folder-description' ('names ' + $cnt + ' files of this folder : ' + ($exs -join ', ')) 'strong'; return }
            }
        }
        function FirstLine { param($gx, [string]$rx)
            foreach ($l0 in $gx.Out) { if ([string]$l0 -match $rx) { return [string]$l0 } }
            return ''
        }
        function InComment { param($no, [int]$ln)
            $arr = GapLines $no
            if (($ln -lt 1) -or ($ln -gt $arr.Count)) { return $false }
            $t0 = $arr[$ln - 1].TrimStart()
            if ($t0 -match '^(//|#|/\*|\*|<!--|'' |rem\s|::|;)') { return $true }
            for ($j0 = $ln - 1; $j0 -ge [Math]::Max(1, $ln - 60); $j0--) {
                $a0 = $arr[$j0 - 1]
                $o0 = $a0.LastIndexOf('/*'); $c0 = $a0.LastIndexOf('*/')
                $o1 = $a0.LastIndexOf('<!--'); $c1 = $a0.LastIndexOf('-->')
                if (($c0 -ge 0) -and ($c0 -gt $o0)) { return $false }
                if (($c1 -ge 0) -and ($c1 -gt $o1)) { return $false }
                if (($o0 -ge 0) -or ($o1 -ge 0)) { return $true }
            }
            return $false
        }
        foreach ($g1 in $gaps) {
            $lnum = 0; if ($g1.Loc -match '^L(\d+)$') { $lnum = [int]$Matches[1] }
            $ext0 = ''; if ($g1.No) { $ext0 = ([string]$fx[[int]$g1.No - 1].File.Extension).ToLowerInvariant() }
            FileRules $g1
            switch ($g1.Kind) {
                'scan-empty' { $h = FirstLine $g1 'direct children incl\. hidden : 0$'; if ($h) { Explain $g1 'empty-folder' $h } }
                'file-read-unexplained' {
                    $tok0 = (([string]$g1.Subject) -replace '\s*\[[a-z\-]+\]\s*$', '')
                    $leaf0 = [SemTool]::BaseName($tok0)
                    $rk = '(?i)LOCALAPPDATA|\bAPPDATA\b|%TEMP%|\bTEMP\b|gettempdir|mkdtemp|tempfile|\.mkdir\s*\(|makedirs|\bruntime\b|_RUNTIME\b|local_root|cache_dir|tmp_dir|work_dir|\buuid'
                    $skip = '(?i)^\s*(from|import)\s|\blog(ger|ging)?\.\w+\s*\(|\bprint\s*\(|\bconsole\.\w+\s*\('
                    $hs = ''; $hw = ''
                    foreach ($l0 in $g1.Out) {
                        $t0 = [string]$l0
                        $m0 = [regex]::Match($t0, '^\s+(def \S+|call \S+|>|) ?L\d+: (.*)$')
                        if (-not $m0.Success) { continue }
                        $code = $m0.Groups[2].Value
                        if (($code -match $skip) -or ($code -notmatch $rk)) { continue }
                        $kd = $m0.Groups[1].Value
                        if ($kd.StartsWith('def ') -or $kd.StartsWith('call ')) { if (-not $hs) { $hs = $t0 } }
                        elseif ($kd -eq '>') { if ((-not $hs) -and ($code.IndexOf($leaf0, [StringComparison]::OrdinalIgnoreCase) -ge 0)) { $hs = $t0 } elseif (-not $hw) { $hw = $t0 } }
                        elseif (-not $hw) { $hw = $t0 }
                    }
                    if ($hs) { Explain $g1 'runtime-path' $hs 'strong' }
                    elseif ($hw) { Explain $g1 'runtime-path-nearby' $hw 'weak' }
                }
                'import-unresolved' {
                    $root = ([string]$g1.Subject).TrimStart('.').Split('.')[0]
                    $h = FirstLine $g1 ('(?i)(pip3?|python\s+-m\s+pip|py\s+-m\s+pip)\s+install\b[^\r\n]*\b' + [regex]::Escape($root) + '\b')
                    if ((-not $h) -and $INSTALLS.ContainsKey($root.ToLowerInvariant())) { $h = $INSTALLS[$root.ToLowerInvariant()] }
                    if ($h) { Explain $g1 'install-instruction' $h }
                    elseif ($BUNDLE.ContainsKey($root.ToLowerInvariant()) -and ($BUNDLE[$root.ToLowerInvariant()] -eq 'flask') -and $flaskUsed) { Explain $g1 'bundled-package' ($root + ' is installed together with flask (flask is imported in this folder)') }
                }
                'isolated-file' {
                    if ($ext0 -eq '.pyc') { Explain $g1 'python-bytecode' 'compiled cache of a .py file (generated by Python)' }
                    else { MentionRule $g1 }
                }
                'role-unclassified' {
                    if ([string]$fx[[int]$g1.No - 1].File.Name -ieq '__init__.py') { Explain $g1 'package-init' ('python package marker; ' + (FirstLine $g1 '^\s+head L2:')) }
                    elseif ($ext0 -in @('.csv', '.tsv')) { Explain $g1 'data-table' ('delimited data file; ' + (FirstLine $g1 '^\s+mention (?!:)')) }
                    elseif ($ext0 -eq '.json') { MentionRule $g1 }
                    elseif ($ext0 -in @('.txt', '.md') -or ($g1.Path -match '\.sample$|\.example')) { $h = FirstLine $g1 '^\s+head L'; if ($h) { Explain $g1 'documentation-or-template' $h } }
                }
                'filename-ambiguous' { if ($lnum -and (InComment $g1.No $lnum)) { Explain $g1 'comment-mention' ('line ' + $g1.Loc + ' is inside a comment') } }
                { $_ -in @('http-ambiguous', 'http-no-route') } {
                    $shp = FirstLine $g1 '^\s+var : property .* also holds'
                    if ($shp -and ($g1.Kind -eq 'http-ambiguous')) { Explain $g1 'shared-property-url' $shp.Trim() 'weak' }
                    if ($g1.Kind -eq 'http-ambiguous') {
                        $ms = @($g1.Out | ForEach-Object { $mq = [regex]::Match([string]$_, '^\s+use L\d+: .*\[method=([A-Z]+)\]'); if ($mq.Success) { $mq.Groups[1].Value } } | Select-Object -Unique)
                        $u1 = FirstLine $g1 '^\s+use L\d+: .*\[method=[A-Z]+\]'
                        if ($ms.Count -ge 2) { Explain $g1 'multi-method-url' ('url is used with ' + ($ms -join ' and ') + ' in this file; ' + $u1.Trim()) 'strong' }
                        elseif ($ms.Count -eq 1) { Explain $g1 'method-from-variable-use' ('method ' + $ms[0] + ' : ' + $u1.Trim()) 'strong' }
                    }
                    $h = FirstLine $g1 '^\s+> L\d+: .*(\b(const|let|var)\s+.*\b[\w$]+\s*=\s*["''`]/|\b[\w$]+\s*:\s*["''`]/)'
                    if ($h -and ($g1.Kind -eq 'http-ambiguous')) { Explain $g1 'url-in-variable' $h }
                    elseif ($h) { $g1.Note = (($g1.Note + ' url is held in a variable / object property; route not found').Trim()) }
                    if (-not $g1.Explained -and ($g1.Kind -eq 'http-no-route')) {
                        $dyn = FirstLine $g1 '^\s+routing .*add_url_rule\s*\(\s*(?![rfu]?["''])[A-Za-z_]'
                        $lit = FirstLine $g1 '^\s+url literal (?!")'
                        if ($dyn -and $lit) {
                            $fd = ([regex]::Match($dyn, '^\s+routing (.+?) L\d+:')).Groups[1].Value
                            $fl = ([regex]::Match($lit, '^\s+url literal (.+?) L\d+:')).Groups[1].Value
                            if ($fd -and ($fd -eq $fl)) { Explain $g1 'dynamic-route-registration' ($dyn.Trim() + ' | ' + $lit.Trim()) 'strong' }
                            else { Explain $g1 'dynamic-route-indirect' ($dyn.Trim() + ' | ' + $lit.Trim()) 'weak' }
                        }
                    }
                }
                'format-not-parsed' {
                    if (FirstLine $g1 'magic=SQLite') { Explain $g1 'sqlite-schema' 'SQLite file; CREATE TABLE statements are extracted in [16-5]' }
                    elseif (FirstLine $g1 'head16=4D 5A') { Explain $g1 'windows-executable' (FirstLine $g1 'head16=4D 5A') }
                    elseif (FirstLine $g1 'Standard (ACE|Jet) DB') { Explain $g1 'access-database' (FirstLine $g1 'Standard (ACE|Jet) DB') }
                    elseif ($ext0 -eq '.pyc') { Explain $g1 'python-bytecode' 'compiled cache of a .py file' }
                    elseif ($SIG.ContainsKey($ext0) -and ($SIG[$ext0].Count -eq 1)) {
                        $sg = @($SIG[$ext0].Keys)[0]; $nf1 = [int]$SIG[$ext0][$sg]
                        if ($nf1 -ge 2) {
                            $asc = -join (($sg -split ' ') | ForEach-Object { $v1 = [Convert]::ToInt32($_, 16); if (($v1 -ge 32) -and ($v1 -lt 127)) { [char]$v1 } else { '.' } })
                            Explain $g1 'consistent-signature' ('all ' + $nf1 + ' ' + $ext0 + ' files start with ' + $sg + ' (' + $asc + ')')
                        }
                    }
                    else {
                        $sec = @($g1.Out | Where-Object { [string]$_ -match '^\s+head L\d+: \[[^\]]+\]\s*$' }).Count
                        $kv = @($g1.Out | Where-Object { [string]$_ -match '^\s+head L\d+: [A-Za-z_][\w\.]*\s*=' }).Count
                        if (($sec -ge 1) -and ($kv -ge 1)) { Explain $g1 'ini-format' ('[section] + key=value lines : sections=' + $sec + ' keys=' + $kv) }
                    }
                }
            }
        }
        $nExp = @($gaps | Where-Object { $_.Explained }).Count
        $nWeak = @($gaps | Where-Object { $_.Explained -and ($_.Conf -eq 'weak') }).Count
        $nStrong = $nExp - $nWeak
        # ---- resolved gaps ----
        $resolved = New-Object System.Collections.ArrayList
        foreach ($k1 in $ledMine.Keys) { $p0 = $ledMine[$k1]; if (($p0.Status -in @('open', 'explained', 'explained-weak')) -and (-not $gapSeen.ContainsKey($k1))) { [void]$resolved.Add($p0) } }
        # ---- ledger write ----
        try {
            $sbl = New-Object System.Text.StringBuilder
            [void]$sbl.Append("Target`tGapKey`tKind`tFirstSeen`tLastSeen`tSeenCount`tStatus`tPath`tSubject`r`n")
            foreach ($ln0 in $ledOther) { [void]$sbl.Append($ln0 + "`r`n") }
            foreach ($g1 in $gaps) { $st1 = 'open'; if ($g1.Explained) { $st1 = 'explained'; if ($g1.Conf -eq 'weak') { $st1 = 'explained-weak' } }; [void]$sbl.Append(($SCANROOT, $g1.Key, $g1.Kind, $g1.First, $RUNID, [string]$g1.Seen, $st1, $g1.Path, $g1.Subject) -join "`t"); [void]$sbl.Append("`r`n") }
            foreach ($k1 in $ledMine.Keys) {
                if ($gapSeen.ContainsKey($k1)) { continue }
                $p0 = $ledMine[$k1]; $lst = $p0.Last; $st0 = $p0.Status
                if ($st0 -in @('open', 'explained', 'explained-weak')) { $lst = $RUNID; $st0 = 'resolved' }
                [void]$sbl.Append(($SCANROOT, $k1, $p0.Kind, $p0.First, $lst, [string]$p0.Seen, $st0, $p0.Path, $p0.Subject) -join "`t"); [void]$sbl.Append("`r`n")
            }
            [IO.File]::WriteAllText($LEDGER, $sbl.ToString(), (New-Object System.Text.UTF8Encoding($false)))
        } catch { Write-Host ('  [warn] gap ledger write: ' + $_.Exception.Message) }
        # ---- assemble gap report ----
        $nNew = @($gaps | Where-Object { $_.Status -in @('new', 'reopened') }).Count
        $nPer = @($gaps | Where-Object { $_.Status -eq 'persisting' }).Count
        $prevTxt = '(no earlier scan of this target)'; if ($ledPrevRun) { $prevTxt = $ledPrevRun } elseif ($prevRun) { $prevTxt = $prevRun + ' (history)' }
        G ('=' * 60)
        G (' folderscan gap report ' + $VER)
        G (' Target    : ' + $SCANROOT)
        G (' Scan time : ' + $RUNID + '   elapsed ' + $MET['elapsed.sec'] + ' s')
        G (' Full scan : ' + $OUT)
        G (' Previous  : ' + $prevTxt)
        G (' Gaps      : ' + $gaps.Count + '  (new ' + $nNew + ' / persisting ' + $nPer + ' / resolved since previous ' + $resolved.Count + ')')
        G (' Explained : strong ' + $nStrong + ' / weak ' + $nWeak + '   Still open : ' + ($gaps.Count - $nExp))
        G '   strong = evidence on the gap line or a traced definition ; weak = nearby / name-only (listed in [G-0b], probed deeper next scan)'
        G ('=' * 60)
        G ' gap   = information this scan could not establish'
        G ' probe = extra evidence collected now for the gap'
        G ' A gap seen again next time is probed deeper (context +3 lines, +1 definition hop, more mentions).'
        G (' http calls whose method was taken from the only matching route (not a gap) : ' + $routeImplied)
        G
        G '---- [G-0] EXPLAINED BY PROBE EVIDENCE (rule hit count / items) ----'
        $xRows = New-Object System.Collections.ArrayList
        foreach ($k0 in $EXR.Keys) { [void]$xRows.Add([string[]]@($k0, [string]$EXC[$k0], [string]$EXR[$k0])) }
        if ($xRows.Count -eq 0) { G '(no gap explained by a rule)' }
        GT 'explain_rules' @('Rule', 'Confidence', 'Hits') $xRows
        $xiRows = New-Object System.Collections.ArrayList
        foreach ($g1 in $gaps) { if ($g1.Explained -and ($g1.Conf -ne 'weak')) { [void]$xiRows.Add([string[]]@($g1.Explained, $g1.Kind, $g1.Path, $g1.Loc, $g1.Subject, $g1.ExEv)) } }
        GT 'explained_gaps' @('Rule', 'Kind', 'Path', 'Loc', 'Subject', 'Evidence') $xiRows
        G
        G '---- [G-0b] WEAKLY EXPLAINED : NEEDS CONFIRMATION (full probe in [G-4], probed deeper next scan) ----'
        $wkRows = New-Object System.Collections.ArrayList
        foreach ($g1 in $gaps) { if ($g1.Explained -and ($g1.Conf -eq 'weak')) { [void]$wkRows.Add([string[]]@($g1.Explained, $g1.Kind, $g1.Path, $g1.Loc, $g1.Subject, $g1.ExEv)) } }
        if ($wkRows.Count -eq 0) { G '(none)' }
        GT 'weakly_explained_gaps' @('Rule', 'Kind', 'Path', 'Loc', 'Subject', 'Evidence') $wkRows
        G
        G '---- [G-1] GAPS BY KIND ----'
        $kRows = New-Object System.Collections.ArrayList
        foreach ($grp in ($gaps | Group-Object Kind | Sort-Object Name)) {
            $nn = @($grp.Group | Where-Object { $_.Status -ne 'persisting' }).Count
            $pv0 = @($resolved | Where-Object { $_.Kind -eq $grp.Name }).Count
            [void]$kRows.Add([string[]]@($grp.Name, [string]@($grp.Group | Where-Object { -not $_.Explained }).Count, [string]$nn, [string]($grp.Count - $nn), [string]$pv0, [string]@($grp.Group | Where-Object { $_.Probed }).Count, [string]@($grp.Group | Where-Object { $_.Explained -and ($_.Conf -ne 'weak') }).Count, [string]@($grp.Group | Where-Object { $_.Explained -and ($_.Conf -eq 'weak') }).Count))
        }
        foreach ($grp in ($resolved | Group-Object Kind)) { if (-not ($gaps | Where-Object { $_.Kind -eq $grp.Name })) { [void]$kRows.Add([string[]]@($grp.Name, '0', '0', '0', [string]$grp.Count, '0', '0', '0')) } }
        if ($kRows.Count -eq 0) { G '(no gap)' }
        GT 'gap_kinds' @('Kind', 'Open', 'New', 'Persisting', 'ResolvedSincePrevious', 'Probed', 'ExplainedStrong', 'ExplainedWeak') $kRows
        G
        G '---- [G-2] GAP LIST ----'
        $lRows = New-Object System.Collections.ArrayList
        foreach ($g1 in $gaps) { $stx = $g1.Status; if ($g1.Explained) { $stx = 'explained:' + $g1.Explained; if ($g1.Conf -eq 'weak') { $stx = 'explained-weak:' + $g1.Explained } }; [void]$lRows.Add([string[]]@($stx, [string]$g1.Seen, $g1.Kind, $g1.Path, $g1.Loc, $g1.Subject, $g1.Note, [string]$g1.Level)) }
        GT 'gaps' @('Status', 'SeenCount', 'Kind', 'Path', 'Loc', 'Subject', 'Note', 'ProbeLevel') $lRows
        G
        G '---- [G-3] RESOLVED SINCE PREVIOUS SCAN ----'
        $vRows = New-Object System.Collections.ArrayList
        foreach ($p0 in $resolved) { [void]$vRows.Add([string[]]@($p0.Kind, $p0.Path, $p0.Subject, $p0.First, [string]$p0.Seen)) }
        if ($vRows.Count -eq 0) { G '(none)' }
        GT 'gaps_resolved' @('Kind', 'Path', 'Subject', 'FirstSeen', 'SeenCount') $vRows
        G
        G ('---- [G-4] PROBES (max ' + $GAPMAX + ' gaps per kind) ----')
        foreach ($g1 in $gaps) {
            if (-not $g1.Probed) { continue }
            if ($g1.Explained -and ($g1.Conf -ne 'weak')) { G ('### [' + $g1.Kind + '] ' + $g1.Path + ' ' + $g1.Loc + '  ' + $g1.Subject + '   EXPLAINED (strong) by ' + $g1.Explained); G ('  evidence : ' + $g1.ExEv); continue }
            if ($g1.Explained) { G ('### [' + $g1.Kind + '] ' + $g1.Path + ' ' + $g1.Loc + '  ' + $g1.Subject + '   EXPLAINED (weak) by ' + $g1.Explained + ' - needs confirmation   (seen ' + $g1.Seen + ', level ' + $g1.Level + ')'); G ('  evidence : ' + $g1.ExEv); foreach ($s2 in $g1.Out) { G $s2 }; continue }
            G ('### [' + $g1.Kind + '] ' + $g1.Path + ' ' + $g1.Loc + '  ' + $g1.Subject + '   (' + $g1.Status + ', seen ' + $g1.Seen + ', level ' + $g1.Level + ')')
            if ($g1.Note) { G ('  note : ' + $g1.Note) }
            foreach ($s2 in $g1.Out) { G $s2 }
            if ($g1.Out.Count -eq 0) { G '  (no probe output)' }
        }
        G
        G '---- [G-5] METRIC CHANGE FROM PREVIOUS SCAN OF THE SAME TARGET ----'
        if (-not $prevRun) { G '(no previous scan of this target in folderscan_history.tsv)' } elseif ($dRows.Count -eq 0) { G '(no change)' }
        GT 'metric_delta' @('Metric', 'Previous', 'Now', 'Delta') $dRows
        G
        # ---- write : report section [19], gap file, batch file ----
        L '==== [19] GAP PROBE (also saved as a separate gaps file) ===='
        foreach ($s0 in $GO) { L $s0 }
        L
        $GAPF = Join-Path (Split-Path $OUT -Parent) ((Split-Path $OUT -Leaf) -replace '^scan_', 'gaps_')
        if ($GAPF -eq $OUT) { $GAPF = $OUT + '.gaps.txt' }
        try {
            [IO.File]::WriteAllLines($GAPF, $GO, $ENC)
            [IO.File]::WriteAllText((Join-Path $WORK 'digestpath.txt'), $GAPF, [Text.Encoding]::Default)
        } catch { Write-Host ('  [warn] gap file write: ' + $_.Exception.Message) }
        if ($env:BATCHDIGEST) {
            try { [IO.File]::AppendAllText($env:BATCHDIGEST, (($GO -join "`r`n") + "`r`n`r`n"), $ENC) } catch { Write-Host ('  [warn] batch gaps: ' + $_.Exception.Message) }
        }
    }
    L '==== END OF REPORT ===='
} catch {
    L ('[FATAL] ' + $_.Exception.Message)
    Write-Host ('[FATAL] ' + $_.Exception.Message)
} finally {
    if ($REPORT_WRITER) { $REPORT_WRITER.Close() }
}
